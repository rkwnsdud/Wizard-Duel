using System;
using System.IO;
using System.Collections;
using System.Collections.Generic;
using System.Reflection;
using UnityEngine;
using Unity.Netcode;
public sealed class NoticeSmokeDriver : MonoBehaviour
{
 string role, dir; NetworkManager net; ArenaWeaponSpawner spawn;
 static object Read(object o,string n)=>o.GetType().GetField(n,BindingFlags.Instance|BindingFlags.NonPublic).GetValue(o);
 static void Write(object o,string n,object v)=>o.GetType().GetField(n,BindingFlags.Instance|BindingFlags.NonPublic).SetValue(o,v);
 string PathFor(string n)=>Path.Combine(dir,n);
 bool Has(string n)=>File.Exists(PathFor(n));
 void Mark(string n,string value="ok") { File.WriteAllText(PathFor(n),value); Debug.Log("SMOKE "+role+" "+n+" "+value); }
 [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)] static void Boot() {
 var a=Environment.GetCommandLineArgs(); if(Array.IndexOf(a,"-spawnSmoke")<0)return;
 var d=new GameObject("Temporary smoke driver").AddComponent<NoticeSmokeDriver>();
 d.role=a[Array.IndexOf(a,"-spawnSmoke")+1]; d.dir=a[Array.IndexOf(a,"-smokeDir")+1];
 Write(FindFirstObjectByType<LobbyNetwork>(),"port",(ushort)17779); Application.runInBackground=true; DontDestroyOnLoad(d); d.StartCoroutine(d.Run());
 }
 void Update(){if(Time.realtimeSinceStartup>210){Mark(role+".timeout");Application.Quit(2);}}
 IEnumerator Run(){
 yield return new WaitForSeconds(2);
 net=NetworkManager.Singleton;
 var lobby=FindFirstObjectByType<LobbyNetwork>();
 lobby.Failed+=s=>Mark(role+".error",s);
 if(role=="a") {
 lobby.SnapshotChanged+=s=>{Mark("code",s.Code);if(s.OpponentPresent&&s.Phase==LobbyPhase.Lobby)lobby.RequestStart();};
 lobby.Create(new DuelSettings{Map=new MapGenerationSettings{Seed=7301,Shape=MapShape.Rectangle,ObstacleDensity=.15f,BushDensity=.05f}});
 } else if(role=="b") {while(!Has("code"))yield return null;lobby.Join(File.ReadAllText(PathFor("code")));}
 while((spawn=FindFirstObjectByType<ArenaWeaponSpawner>())==null||!spawn.Running)yield return null;
 if(role=="server") {
 var round=(WeaponSpawnRound)Read(spawn,"round");
 // Deterministic weapon choice for the integration scenario; production catalog is unchanged.
 Write(round,"weapons",new byte[]{3});
 double began=net.ServerTime.Time;
 while(round.Drops.Count==0)yield return null;
 Mark("first",(net.ServerTime.Time-began).ToString("F2",System.Globalization.CultureInfo.InvariantCulture));
 while(!Has("a.swapped")||!Has("b.swapped"))yield return null;
 Write(round,"weapons",new byte[]{4});
 while(round.Drops.Count==0)yield return null;
 Mark("second");
 while(!Has("a.second")||!Has("b.second"))yield return null;
 FindFirstObjectByType<DuelArena>().EndCombat();
 yield return new WaitForSeconds(2);
 if(round.Running||round.Drops.Count!=0){Mark("server.error","cleanup");yield break;}
 Mark("server.pass");
 while(!Has("a.pass")||!Has("b.pass"))yield return null;
 Application.Quit();yield break;
 }
 if(Read(spawn,"notification")!=null){Mark(role+".error","notification before spawn");yield break;}
 var player=net.LocalClient.PlayerObject;
 var inv=player.GetComponent<PlayerWeaponInventory>();int events=0;inv.StoredWeaponChanged+=_=>events++;
 var drops=(Dictionary<uint,SpawnedWeapon>)Read(spawn,"clientDrops");
 while(drops.Count==0)yield return null;
  var notice=(WeaponSpawnNotification)Read(spawn,"notification");
 if(notice==null||!notice.enabled){Mark(role+".error","missing first notification");yield break;}
 float firstNotice=(float)Read(notice,"startedAt");
 var source=(AudioSource)Read(notice,"audioSource");
 var group=(CanvasGroup)Read(notice,"group");
 var labelNotice=group.GetComponentInChildren<TMPro.TextMeshProUGUI>();
 if(labelNotice.text!="마법 무기가 등장했습니다!"||group.blocksRaycasts||source.clip==null||source.spatialBlend!=0||Mathf.Abs(source.volume-.22f)>.001f||!source.isPlaying){Mark(role+".error","notification text/audio/input");yield break;}
 Mark(role+".firstNotice","audio-playing volume="+source.volume+" duration="+source.clip.length);
 StartCoroutine(CheckFade(notice,firstNotice));
 SpawnedWeapon target=default;foreach(var d in drops.Values){target=d;break;}
 Mark(role+".drop",target.Id+":"+target.Weapon+":"+target.Ground);
 if(target.Weapon!=3){Mark(role+".error","weapon");yield break;}
 // Outside pickup range must not alter inventory.
 inv.TryPickupNearby();yield return new WaitForSeconds(.3f);
 if(inv.StoredWeapon!=null){Mark(role+".error","distance");yield break;}
 player.transform.position=target.Ground+new Vector3(role=="a"?.6f:-.6f,.2f,0);
 yield return new WaitForSeconds(1.5f);Mark(role+".ready");
 while(!Has("a.ready")||!Has("b.ready"))yield return null;
 inv.TryPickupNearby();
 yield return new WaitForSeconds(2);
 if(drops.Count!=0||spawn.RequestPending){Mark(role+".error","pickup/remove/ack");yield break;}
 bool winner=inv.StoredWeapon!=null;
 Mark(role+".claim",winner?"winner":"loser");
 if(winner){
 var hud=player.GetComponent<SecondaryWeaponHud>();
 var label=(TMPro.TextMeshProUGUI)Read(hud,"nameText");
 if(label.text!="Light Orb"||events!=1){Mark(role+".error","HUD/event");yield break;}
 ScreenCapture.CaptureScreenshot(PathFor(role+"-hud.png"));
 inv.TrySwapWeapon();
 }
 yield return new WaitForSeconds(2);
 if(winner && (!(player.GetComponent<PlayerAttack>().CurrentWeapon is LightOrbWeapon)||events!=2||spawn.RequestPending)){
 Mark(role+".error","swap");yield break;}
 int owners=0,orbs=0,stored=0;
 foreach(var p in FindObjectsByType<DuelPlayer>(FindObjectsSortMode.None)){
 if(Read(p.GetComponent<SecondaryWeaponHud>(),"canvasObject")!=null)owners++;
 if(p.GetComponent<PlayerAttack>().CurrentWeapon is LightOrbWeapon)orbs++;
 if(p.GetComponent<PlayerWeaponInventory>().StoredWeapon!=null)stored++;
 }
 if(owners!=1||orbs!=1||stored!=1){Mark(role+".error","replicas/local HUD "+owners+"/"+orbs+"/"+stored);yield break;}
 if((float)Read(notice,"startedAt")!=firstNotice){Mark(role+".error","inventory snapshot replayed notice");yield break;}
 Mark(role+".swapped");
  while(drops.Count==0)yield return null;
 float secondNotice=(float)Read(notice,"startedAt");
 if(!notice.enabled||!source.isPlaying||secondNotice-firstNotice<29.5f||secondNotice-firstNotice>31f){Mark(role+".error","second interval notification");yield break;}
 Mark(role+".second","notice interval="+(secondNotice-firstNotice));
 while(spawn.Running)yield return null;
 yield return new WaitForSeconds(.5f);
 if(notice.enabled||source.isPlaying||group.gameObject.activeSelf||!Has(role+".fade")||drops.Count!=0||FindObjectsByType<FloatingWeaponPickup>(FindObjectsSortMode.None).Length!=0){Mark(role+".error","cleanup");yield break;}
 Mark(role+".pass");yield return new WaitForSeconds(2);Application.Quit();
 }
 IEnumerator CheckFade(WeaponSpawnNotification notice,float began){
 var group=(CanvasGroup)Read(notice,"group");
 while(Time.unscaledTime-began<.3f)yield return null;
 if(group.alpha<.99f){Mark(role+".error","fade-in");yield break;}
 while(Time.unscaledTime-began<2.32f)yield return null;
 if(group.alpha<=0||group.alpha>=1){Mark(role+".error","fade-out");yield break;}
 while(Time.unscaledTime-began<2.8f)yield return null;
 if(notice.enabled||group.gameObject.activeSelf){Mark(role+".error","notice remained visible");yield break;}
 Mark(role+".fade");
 }}
